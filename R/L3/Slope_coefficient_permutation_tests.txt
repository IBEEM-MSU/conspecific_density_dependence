Demography_data <- read.csv("C:/Users/eagar/Dropbox/IBEEM/conspecific_density_dependence/data/L3/Combined_demography_modeling_data.csv")

nperm <- 999 # To test this code, drop this number to 9 or 99 - at 999 permutations; it can take about 20-30 minutes to compile on a PC with an i7 core and 64 gb RAM at 999 for certain species.
# Analyses are divided first by species, then by analysis type. Test A, Test B, Test C, and Test D correspond to the four panels of Fig. 3 in Eagar et al. 2024. Test A tests for conspecific density dependence based on seedling mycorrhizal type. Test B tests for density dependent effects related to conspecific density in interaction with the surrounding tree community's mycorrhizal dominance. Test C tests for density dependent effects related to conmycorrhizal density in interaciton with the surrounding tree community's mycorrhizal dominance. Test D tests a new hypothesis concerning density dependent effects structred by seedling mycorrhizal types in con- and heteromycorrhizal dominant communities.

# Each of the four tests are done with both raw basal area values and proportional values.

# Permutation test for all species together

Rand_results_con_den <- matrix(nrow = nperm, ncol = 1, 0) # We have 8 terms!!! to test in this model, so we'll need 8 dataframes to store the randomized F-value test results.
	Rand_results_seedling_myc <- matrix(nrow = nperm, ncol = 1, 0)
	Rand_results_AM_den <- matrix(nrow = nperm, ncol = 1, 0)
	Rand_results_ECM_den <- matrix(nrow = nperm, ncol = 1, 0)
	Rand_results_con_den_seedling_myc_AM_den <- matrix(nrow = nperm, ncol = 1, 0)
	Rand_results_con_den_seedling_myc_ECM_den <- matrix(nrow = nperm, ncol = 1, 0)	
	Rand_results_con_seedlings <- matrix(nrow = nperm, ncol = 1, 0)
	Rand_results_other_seedlings <- matrix(nrow = nperm, ncol = 1, 0)

for(i in 1:nperm) # For loop for randomized F-values
{

	All_sp_slope_modeling_data <- Demography_data %>%
					group_by(Cohort.Year, SiteID) %>%
					mutate(Rand.GLM.Slope.Estimate = sample(GLM.Slope.Estimate, n())) # This randomizes the slope coefficients for each quadrat within a site and cohort year without replacing values.

		# Test_data <- All_sp_slope_modeling_data %>%
							# filter(Cohort.Year == 2002, SiteID == 3) %>%
							# select(GLM.Slope.Estimate, Rand.GLM.Slope.Estimate)
			
			# as.data.frame(Test_data)
	
	All_sp_mod_BA_rand <- lmer(Rand.GLM.Slope.Estimate ~ consp_15m + Seedling_myc + 
                       AM_15m + ECM_15m + 
                       consp_15m:Seedling_myc:AM_15m + # Panel D
                       consp_15m:Seedling_myc:ECM_15m + # Panel D
                       log0yr + logPrior +
                       (1|SiteID) + (1|Cohort.Year) + (1|Sp), 
					   data = All_sp_slope_modeling_data)
					
	Rand_aov <- anova(All_sp_mod_BA_rand) # Save results to store F-values below
					
		Rand_results_con_den[i] <- Rand_aov$"F value"[1]
		Rand_results_seedling_myc[i] <-  Rand_aov$"F value"[2]
		Rand_results_AM_den[i] <- Rand_aov$"F value"[3]
		Rand_results_ECM_den[i] <- Rand_aov$"F value"[4]
		Rand_results_con_seedlings[i] <- Rand_aov$"F value"[5]
		Rand_results_other_seedlings[i] <- Rand_aov$"F value"[6]
		Rand_results_con_den_seedling_myc_AM_den[i] <- Rand_aov$"F value"[7]
		Rand_results_con_den_seedling_myc_ECM_den[i] <- Rand_aov$"F value"[8]
					
}

	# LM test with real data. We will compare these F-values to those from the peAll_sputations above by ranking their magnitude. If our observations are not statistically significant, the true F-values will rank somewhere within the "normal" simulated distribution of F-values. If our observations are significant, our F-values will fall towards one of the tails of the distribution and a critical threshold (p-value at 0.05) can be estimated.
			
	All_sp_mod_BA_true <- lmer(GLM.Slope.Estimate ~ consp_15m + Seedling_myc + 
                       AM_15m + ECM_15m + 
                       consp_15m:Seedling_myc:AM_15m + # Panel D
                       consp_15m:Seedling_myc:ECM_15m + # Panel D
                       log0yr + logPrior +
                       (1|SiteID) + (1|Cohort.Year) + (1|Sp), 
					   data = All_sp_slope_modeling_data)
					
	Obs_aov <- anova(All_sp_mod_BA_true)

		Obs_results_con_den <- as.vector(Obs_aov[1,5])
			All_results_con_den <- rbind(Rand_results_con_den, Obs_results_con_den)
			Ranks_con_den <- rank(All_results_con_den)
			P_values_con_den <- 1-Ranks_con_den[nperm+1]/(nperm+1)

		Obs_results_seedling_myc <- as.vector(Obs_aov[2,5])
			All_results_seedling_myc <- rbind(Rand_results_seedling_myc, Obs_results_seedling_myc)
			Ranks_seedling_myc <- rank(All_results_seedling_myc)
			P_values_seedling_myc <- 1-Ranks_seedling_myc[nperm+1]/(nperm+1)
			
		Obs_results_AM_den <- as.vector(Obs_aov[3,5])
			All_results_AM_den <- rbind(Rand_results_AM_den, Obs_results_AM_den)
			Ranks_AM_den <- rank(All_results_AM_den)
			P_values_AM_den <- 1-Ranks_AM_den[nperm+1]/(nperm+1)
			
		Obs_results_ECM_den <- as.vector(Obs_aov[4,5])
			All_results_ECM_den <- rbind(Rand_results_ECM_den, Obs_results_ECM_den)
			Ranks_ECM_den <- rank(All_results_ECM_den)
			P_values_ECM_den <- 1-Ranks_ECM_den[nperm+1]/(nperm+1)
		
		Obs_results_con_seedlings <- as.vector(Obs_aov[5,5])
			All_results_con_seedlings <- rbind(Rand_results_con_seedlings, Obs_results_con_seedlings)
			Ranks_con_seedlings <- rank(All_results_con_seedlings)
			P_values_con_seedlings <- 1-Ranks_con_seedlings[nperm+1]/(nperm+1)
			
		Obs_results_other_seedlings <- as.vector(Obs_aov[6,5])
			All_results_other_seedlings <- rbind(Rand_results_other_seedlings, Obs_results_other_seedlings)
			Ranks_other_seedlings <- rank(All_results_other_seedlings)
			P_values_other_seedlings <- 1-Ranks_other_seedlings[nperm+1]/(nperm+1)
			
		Obs_results_con_den_seedling_myc_AM_den <- as.vector(Obs_aov[7,5])
			All_results_con_den_seedling_myc_AM_den <- rbind(Rand_results_con_den_seedling_myc_AM_den, Obs_results_con_den_seedling_myc_AM_den)
			Ranks_con_den_seedling_myc_AM_den <- rank(All_results_con_den_seedling_myc_AM_den)
			P_values_con_den_seedling_myc_AM_den <- 1-Ranks_con_den_seedling_myc_AM_den[nperm+1]/(nperm+1)	
			
		Obs_results_con_den_seedling_myc_ECM_den <- as.vector(Obs_aov[8,5])
			All_results_con_den_seedling_myc_ECM_den <- rbind(Rand_results_con_den_seedling_myc_ECM_den, Obs_results_con_den_seedling_myc_ECM_den)
			Ranks_con_den_seedling_myc_ECM_den <- rank(All_results_con_den_seedling_myc_ECM_den)
			P_values_con_den_seedling_myc_ECM_den <- 1-Ranks_con_den_seedling_myc_ECM_den[nperm+1]/(nperm+1)
			
	P_values_con_den
		hist(All_results_con_den) # vs
		Obs_results_con_den
	
	P_values_seedling_myc
		hist(All_results_seedling_myc) # vs
		Obs_results_seedling_myc
		
	P_values_AM_den
		hist(All_results_AM_den) # vs
		Obs_results_AM_den
		
	P_values_ECM_den
		hist(All_results_ECM_den) # vs
		Obs_results_ECM_den

	P_values_con_seedlings
		hist(All_results_con_seedlings)
		Obs_results_con_seedlings
		
	P_values_other_seedlings
		hist(All_results_other_seedlings)
		Obs_results_other_seedlings
		
	P_values_con_den_seedling_myc_AM_den
		hist(All_results_con_den_seedling_myc_AM_den)
		Obs_results_con_den_seedling_myc_AM_den

	P_values_con_den_seedling_myc_ECM_den
		hist(All_results_con_den_seedling_myc_ECM_den)
		Obs_results_con_den_seedling_myc_ECM_den
	
	r.squaredGLMM(All_sp_mod_BA_true)
	
	# Store modeling results & export for plotting.

	summary(All_sp_mod_BA_true)
	
	All_sp_GLM_slope_coefficient_modeling_results <- data.frame(species = c("All_sp", "All_sp", "All_sp", "All_sp", "All_sp", "All_sp", "All_sp", "All_sp", "All_sp", "All_sp"), 
															term = c("Conspecific_BA", "Seedling_myc", "AM_BA", "ECM_BA", "Log_Recruit_Density", "Log_Prior_Density", "Conspecific_BA_seedling_AM_AM_den", "Conspecific_BA_seedling_ECM_AM_den","Conspecific_BA_seedling_AM_ECM_den", "Conspecific_BA_seedling_ECM_ECM_den"), 
															estimate = c("-1.5767", "0.4882", "-1.0172", "0.7403", "19.3914", "1.9193", "-0.3364", "-0.3274", "0.4740", "-0.3171"), 
															std.error = c("1.1145", "3.2490", "0.5025", "0.3042", "0.2688", "0.3452", "0.6170", "0.7484", "0.3971", "0.3827"), 
															t_value = c("-1.415", "0.150", "-2.025", "2.433", "72.152", "5.560", "-0.545", "-0.438", "1.194", "-0.829"), 
															p.value = c(P_values_con_den, P_values_seedling_myc, P_values_AM_den, P_values_ECM_den, P_values_con_seedlings, P_values_other_seedlings, P_values_con_den_seedling_myc_AM_den, P_values_con_den_seedling_myc_AM_den, P_values_con_den_seedling_myc_ECM_den, P_values_con_den_seedling_myc_ECM_den),
															r.squared = c("0.1390524", "0.1390524", "0.1390524", "0.1390524", "0.1390524", "0.1390524", "0.1390524", "0.1390524", "0.1390524", "0.1390524"))
