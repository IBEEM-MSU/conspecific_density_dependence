					gather(c(11:36), key = "Obs.Year", value = "Perc") %>%
					mutate(Time = 1 + (LY - FY)) %>% # Time since original cohort recording. # unique(rm_stats[,c(9, 10, 13, 14)])
					filter(Perc >= 0.05) %>% # Remove times where 5% or fewer of seedlings remain. This allows us to approxiamte a "time to 95% mortality" point. # plot(rm_stats$Time, rm_stats$Perc)
					group_by(First.Year) %>%
					mutate(Mean = mean(Time), Median = median(Time)) %>% # Calculate mean and median time to 95% mortality for each cohort.
					ungroup() %>%
					mutate(Mean_overall = mean(Time), Median_overall = median(Time)) %>% # Calculate mean and median time to 95% mortality for all cohorts combined.
					select(First.Year, Mean, Median, Mean_overall, Median_overall) %>%
					unique()

rm_16_35 <- read.csv("C:/Users/eagar/Dropbox/IBEEM/conspecific_density_dependence/data/L0/rm_16_35.csv") %>%
					replace(is.na(.), 0) 
# for(each year):					
Year_1998 <- rm_16_35 %>%
					gather(c(11:36), key = "Obs.Year", value = "Count") %>%
					filter(First.Year == "1998") %>%
					separate(Obs.Year, into = c("X", "Obs.Year"), sep = "X") %>%
					mutate(Time = 1 + (as.numeric(Obs.Year) - as.numeric(First.Year))) %>%
					filter(Time <= 4)
					
	LM_1998 <- lm(Count ~ Time, data = Year_1998)

	Slope_coeff <- as.vector(LM_1998$coefficients)

	Slope_coeff[2]
# for(each quadrat):
Year_1998_quad_1 <- Year_1998 %>%
					filter(quadrat == 1)
					
	LM_1998_quad_1 <- lm(Count ~ Time, data = Year_1998_quad_1)

	Slope_coeff_quad_1 <- as.vector(LM_1998_quad_1$coefficients)

	Slope_coeff_quad_1[2]

	log(Slope_coeff[2]/Slope_coeff_quad_1[2]) # Log ratio of the plot level slope vs quadrat slope. This tells us if the individual quadrat slope is more negative than overall (- values) or less negative overall (+ values).
