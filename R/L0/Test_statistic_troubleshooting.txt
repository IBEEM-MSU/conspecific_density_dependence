# Write a compare and replace function:
compare_replace <- function(df) {
  # Loop through all columns except the last one
  for (i in (ncol(df) - 1):1) {
    # Compare each value in column i to the value in the next column (i+1)
    df[[i]] <- ifelse(df[[i]] < df[[i + 1]], df[[i + 1]], df[[i]])
  }
  return(df)
}

	Red_maple_demography[,c(12:38)] <- compare_replace(Red_maple_demography[,c(12:38)]) # Correct for seedlings missed in a previous year's census
