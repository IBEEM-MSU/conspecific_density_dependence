# Obstacle 3: Obtain slope coefficients for each quadrat. Modeling done via GLM with binomial distribution due to data being proportional in nature 

compare_replace <- function(df) { # Write a compare and replace function. This corrects for missed observations in early cohort years (yr 1, 10 seedlings -> yr 2, 11 seedlings gets corrected to yr 1, 11 seedlings)
  
	for(i in (ncol(df) - 1):1) {
   
		df[[i]] <- ifelse(df[[i]] < df[[i + 1]], df[[i + 1]], df[[i]])
	
	}
  
	return(df)
	
}

American_elm_demography[,c(12:38)] <- compare_replace(American_elm_demography[,c(12:38)]) # Correct for seedlings missed in a previous year's census
					
American_elm_demography_subset <- American_elm_demography %>% # Foaeat data & filter for time (years 0 - 3); remove early cohorts with insufficient data and later cohorts that lack enough time (e.g., cohort 1998 with only 3 years of observations, 2024 cohort)
					filter(sumAll > 0) %>% # Remove quadrats with no observed seedlings at a given time and site
					gather(c(12:38), key = "Obs.Year", value = "Count") %>%
					separate(Obs.Year, into = c("X", "Obs.Year"), sep = "X") %>%
					mutate(Time.Since.First.Year = as.numeric(Obs.Year) - as.numeric(First.Year)) %>%
					filter(Time.Since.First.Year <= 3 & Time.Since.First.Year >= 0) %>% # Filter for time
					filter(First.Year >= 2002 & First.Year <= 2021) %>% # Filter for cohorts
					select(c(2, 4, 5, 7, 14, 15)) # Reduce column number

	write.csv(American_elm_demography_subset, "C:/Users/eagar/Dropbox/IBEEM/conspecific_density_dependence/data/L1/ae_count_demography.csv", row.names = FALSE) # Export data for random slopes modeling

# For loops for obtaining slope estimates from Binomial GLMs
	
Year_results <- matrix(nrow = 0, ncol = 6, 0)
	colnames(Year_results) <- c('TransectID', 'SiteID','quadrat', 'Cohort.Year', 'GLM.Slope.Estimate', 'Cohort.Count') # Create matrix to store slopes by year by site by quadrat

for(y in 2002:2021) {

	By_year <- American_elm_demography_subset %>%
						filter(First.Year == y)
			
	Site_results <- matrix(nrow = 0, ncol = 6, 0)
		colnames(Site_results) <- c('TransectID', 'SiteID','quadrat', 'Cohort.Year', 'GLM.Slope.Estimate', 'Cohort.Count') # Create matrix to store slopes by site by quadrat
	
	for(s in 1:12) {

		By_site <- By_year %>%
							filter(SiteID == s)
		
		Slope_coefficients <- matrix(nrow = 200, ncol = 6, 0) # Matrix to store slopes.nrow(American_elm_demography_subset)
			colnames(Slope_coefficients) <- c('TransectID', 'SiteID','quadrat', 'Cohort.Year', 'GLM.Slope.Estimate', 'Cohort.Count') # Create matrix to store slopes by quadrat
		
		for(q in 1:200) {
		
			By_quadrat <- By_site %>%
								filter(quadrat == q)
			
			if(length(By_quadrat$quadrat) > 0) { # This if-else statement checks to see if a quadrat has data to model. If the quadrat (or above site or year) is devoid of seedlings the object has a length of 0 and fails the check, defailting to the else clause that populates the storage matrix with NAs. If the object does have data, it runs a Binomial GLM and stores the slope estimate along with quadrat, site, and year inforamtion
										
				GLM_quadrat <- glm(cbind(Count, max(By_quadrat$Count) - Count)  ~ Time.Since.First.Year, family = binomial(link = "logit"), data = By_quadrat)
				
				Slope_coefficients[q,1] <- unique(By_quadrat$TransectID)
				Slope_coefficients[q,2] <- unique(By_site$SiteID)
				Slope_coefficients[q,3] <- q
				Slope_coefficients[q,4] <- unique(By_year$First.Year)
				Slope_coefficients[q,5] <- as.vector(GLM_quadrat$coefficients[2])
				Slope_coefficients[q,6] <- max(By_quadrat$Count)
				
			} else {
												
				Slope_coefficients[q,1] <- NA
				Slope_coefficients[q,2] <- s
				Slope_coefficients[q,3] <- q
				Slope_coefficients[q,4] <- y
				Slope_coefficients[q,5] <- NA
				Slope_coefficients[q,6] <- 0
				
			}

		}

		Site_results <- rbind(Site_results, na.omit(Slope_coefficients)) # This adds the results of the current loop to those from previous loops. It also removes those combinations of quadrats, sites, and years where there were no observations to simplify the data by removing NAs
		
	}
	
Year_results <- rbind(Year_results, Site_results)

}

# Export data

Year_results <- as.data.frame(Year_results)

write.csv(Year_results, "C:/Users/eagar/Dropbox/IBEEM/conspecific_density_dependence/data/L1/ae_GLM_slopes.csv", row.names = FALSE)
